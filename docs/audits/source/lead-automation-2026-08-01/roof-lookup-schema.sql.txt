-- Roof-lookup enrichment schema
-- Supports the Google Solar API lookup step gated behind consented lead
-- form submission (see roof-lookup-workflow.json / roof-lookup-workflow-notes.md)

-- 1. Cache of Google Solar API roof lookups, keyed by address so duplicate
--    leads at the same address never trigger a second API call.
create table if not exists roof_lookups (
  id uuid primary key default gen_random_uuid(),

  normalized_address text not null,
  place_id text,
  latitude numeric(9,6),
  longitude numeric(9,6),

  lookup_status text not null check (
    lookup_status in ('success', 'no_coverage', 'error', 'quota_exhausted')
  ),
  api_call_made boolean not null default false,   -- false = served from cache

  raw_response jsonb,          -- full Solar API response, kept for audit/reprocessing
  roof_segments jsonb,         -- [{ pitch_degrees, azimuth_degrees, area_sqft }, ...]
  plane_count int,
  total_footprint_sqft numeric,
  total_adjusted_sqft numeric, -- pitch-corrected surface area
  pitch_source text check (pitch_source in ('api', 'default', 'manual')),
  imagery_date date,
  imagery_quality text,

  quote_range_low numeric,
  quote_range_high numeric,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create unique index if not exists roof_lookups_normalized_address_idx
  on roof_lookups (normalized_address);

create index if not exists roof_lookups_place_id_idx
  on roof_lookups (place_id);

-- keep updated_at current on every upsert
create or replace function set_roof_lookups_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

drop trigger if exists roof_lookups_set_updated_at on roof_lookups;
create trigger roof_lookups_set_updated_at
  before update on roof_lookups
  for each row execute function set_roof_lookups_updated_at();


-- 2. Lightweight internal counter so n8n can self-throttle before Google's
--    free-tier cap (or your GCP quota) gets hit mid-month.
create table if not exists api_usage_counters (
  id uuid primary key default gen_random_uuid(),
  api_name text not null,   -- e.g. 'google_solar_building_insights'
  period text not null,     -- 'YYYY-MM', e.g. '2026-07'
  call_count int not null default 0,
  updated_at timestamptz not null default now(),
  unique (api_name, period)
);

-- atomic increment, callable from n8n via PostgREST RPC:
-- POST {SUPABASE_URL}/rest/v1/rpc/increment_api_usage_counter
-- body: { "p_api_name": "google_solar_building_insights", "p_period": "2026-07" }
create or replace function increment_api_usage_counter(p_api_name text, p_period text)
returns void as $$
  insert into api_usage_counters (api_name, period, call_count)
  values (p_api_name, p_period, 1)
  on conflict (api_name, period)
  do update set call_count = api_usage_counters.call_count + 1,
                updated_at = now();
$$ language sql;


-- 3. Columns to add onto your existing leads table.
-- Adjust the table name below if leads live under a different name in Supabase.
alter table leads
  add column if not exists roof_lookup_id uuid references roof_lookups(id),
  add column if not exists consent_to_contact boolean not null default false,
  add column if not exists quote_range_sent_at timestamptz;
